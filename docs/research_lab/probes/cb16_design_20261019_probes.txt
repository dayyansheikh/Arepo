--- G1 script (R4 text; R3 differs only in the `sp =` line and lacks the final category block) ---
# counts only: grain (third decimal) share and large-print inter-arrival gaps, SCREEN window of CB13-T
import json, collections
LO, HI = 1791763200, 1792108800
files = ["trades-2026101%d.jsonl" % d for d in (2, 3, 4, 5)]
grain = collections.Counter()
last = {}
gaps = collections.Counter()
seen_tx = set()
for f in files:
    with open(f) as fh:
        for line in fh:
            try:
                r = json.loads(line)
            except Exception:
                continue
            ts = r.get("ts")
            if not isinstance(ts, (int, float)) or not (LO <= ts < HI):
                continue
            cat = r.get("ctx_category") or "none"
            sp = "sports" if str(cat).lower().startswith("sport") else ("none" if cat == "none" else "nonsports")
            p = r.get("price")
            try:
                k = round(float(p) * 1000)
            except Exception:
                grain[(sp, "bad")] += 1
                continue
            band = "mid" if 50 <= k <= 950 else "tail"
            grain[(sp, band, "sub" if k % 10 else "whole")] += 1
            cid = r.get("condition_id")
            key = (cid, r.get("tx16"))
            if key in seen_tx:
                continue
            seen_tx.add(key)
            prev = last.get(cid)
            if prev is not None and ts >= prev:
                g = ts - prev
                b = "<1h" if g < 3600 else ("1-6h" if g < 21600 else ("6-12h" if g < 43200 else ">=12h"))
                gaps[(sp, b)] += 1
            elif prev is None:
                gaps[(sp, "first")] += 1
            else:
                gaps[(sp, "out_of_order")] += 1
            last[cid] = max(ts, prev or ts)
for k in sorted(grain):
    print("grain", k, grain[k])
for k in sorted(gaps):
    print("gap", k, gaps[k])
cats = collections.Counter()
for f in files:
    with open(f) as fh:
        for line in fh:
            try:
                r = json.loads(line)
            except Exception:
                continue
            ts = r.get("ts")
            if isinstance(ts, (int, float)) and LO <= ts < HI:
                cats[r.get("ctx_category")] += 1
print("cats", cats.most_common(15))
--- R3 `sp =` line ---
            sp = "sports" if cat == "sports" else ("none" if cat == "none" else "nonsports")

Notes on G1. json.loads decodes every field of a monitor record into memory, including the forbidden `side` and `wallet_*`
fields. The script reads only ts, ctx_category, price, condition_id and tx16, and prints only counts. It reads no
outcome, mark, forward price or later capture. Prices of prints in [2026-10-12, 2026-10-16) are a feature class that CB13
H6-H8 had already disclosed.
